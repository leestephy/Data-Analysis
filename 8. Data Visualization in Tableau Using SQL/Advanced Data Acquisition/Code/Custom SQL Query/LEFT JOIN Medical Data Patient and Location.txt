```sql
SELECT CAST("location_staging2"."city" AS TEXT) AS "city",
  CAST("location_staging2"."county" AS TEXT) AS "county",
  "location_staging2"."location_id" AS "location_id (location_staging2)",
  "patient_subset"."location_id" AS "location_id",
  CAST("patient_subset"."patient_id" AS TEXT) AS "patient_id",
  CAST("patient_subset"."readmis" AS TEXT) AS "readmis",
  CAST("location_staging2"."state" AS TEXT) AS "state",
  CAST("location_staging2"."zipcode" AS TEXT) AS "zipcode"
FROM "public"."patient_subset" "patient_subset"
  LEFT JOIN "public"."location_staging2" "location_staging2" ON ("patient_subset"."location_id" = "location_staging2"."location_id")
```
